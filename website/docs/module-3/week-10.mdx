### Nav2 Architecture for Humanoid Robots

Navigation2 provides a flexible and extensible navigation framework that can be adapted for humanoid robots. The key difference is that bipedal robots require consideration of foot placement, balance, and dynamic walking patterns.

#### Nav2 Core Components

Nav2 consists of several core components that work together to provide navigation capabilities:

- **Global Planner**: Creates a path from the current location to the goal
- **Local Planner**: Executes the path while avoiding obstacles and respecting robot dynamics
- **Controller**: Translates planned motions into actuator commands
- **Costmap**: Represents obstacles and navigable space
- **Transform System**: Manages coordinate frame transformations

        q.x /= norm
        q.y /= norm
        q.z /= norm

        sinr_cosp = 2 * (q.w * q.x + q.y * q.z)
        cosr_cosp = 1 - 2 * (q.x * q.x + q.y * q.y)

        cosy_cosp = 1 - 2 * (q.y * q.y + q.z * q.z)
        yaw = math.atan2(siny_cosp, cosy_cosp)

        return [roll, pitch, yaw]

class BipedalConstraintManager(Node):
    """
    Manages bipedal kinematic constraints for Nav2 path planning
    """

    def __init__(self):
        super().__init__('bipedal_constraint_manager')

        # Bipedal-specific parameters
        self.step_height_limit = 0.15  # meters
        self.step_width_limit = 0.5    # meters
        self.turning_radius = 0.4      # meters
        self.leg_length = 0.8          # meters
        self.support_polygon_margin = 0.1  # meters

        # Publishers and subscribers
        self.path_sub = self.create_subscription(
            Path,
            'global_plan',
            self.path_callback,
            10
        )

        self.constrained_path_pub = self.create_publisher(
            Path,
            'constrained_global_plan',
            10
        )

        # Timer for constraint validation
        self.constraint_timer = self.create_timer(0.1, self.validate_constraints)

        self.get_logger().info("Bipedal constraint manager initialized")
        self.get_logger().info("Importance of Physical AI: This demonstrates how AI navigation systems account for physical robot constraints")

    def path_callback(self, msg):
        """Process global path with bipedal constraints"""
        constrained_path = self.apply_bipedal_constraints(msg)
        self.constrained_path_pub.publish(constrained_path)

    def apply_bipedal_constraints(self, original_path):
        """Apply bipedal kinematic constraints to path"""
        constrained_path = Path()
        constrained_path.header = original_path.header

        # Apply constraints to each point in the path
        for i, pose in enumerate(original_path.poses):
            if i == 0:
                # First pose remains unchanged (current position)
                constrained_path.poses.append(pose)
            else:
                # Calculate constraints for this pose based on previous
                constrained_pose = self.constrain_pose(pose, constrained_path.poses[-1])

                # Only add if pose is valid under constraints
                if self.is_pose_valid(constrained_pose, constrained_path.poses[-1]):
                    constrained_path.poses.append(constrained_pose)

        return constrained_path

    def constrain_pose(self, target_pose, current_pose):
        """Apply bipedal constraints to a target pose"""
        constrained_pose = PoseStamped()
        constrained_pose.header = target_pose.header

        # Calculate the displacement from current pose
        dx = target_pose.pose.position.x - current_pose.pose.position.x
        dy = target_pose.pose.position.y - current_pose.pose.position.y
        distance = math.sqrt(dx*dx + dy*dy)

        # Apply step width limit
        if distance > self.step_width_limit:
            # Scale the displacement to fit within step limit
            scale_factor = self.step_width_limit / distance
            dx *= scale_factor
            dy *= scale_factor

        # Apply turning radius constraint
        # Calculate the angle change from previous movement direction
        if len(self.get_current_path()) > 1:
            prev_dx = current_pose.pose.position.x - self.get_current_path()[-2].pose.position.x
            prev_dy = current_pose.pose.position.y - self.get_current_path()[-2].pose.position.y
            prev_angle = math.atan2(prev_dy, prev_dx)
            target_angle = math.atan2(dy, dx)

            angle_diff = abs(target_angle - prev_angle)
            if angle_diff > math.pi:
                angle_diff = 2 * math.pi - angle_diff

            # Limit turning angle to respect turning radius
            max_turn_angle = self.step_width_limit / self.turning_radius
            if angle_diff > max_turn_angle:
                # Adjust target direction to respect turning constraint
                adjusted_angle = prev_angle + max_turn_angle if angle_diff > 0 else prev_angle - max_turn_angle
                dx = math.cos(adjusted_angle) * min(distance, self.step_width_limit)
                dy = math.sin(adjusted_angle) * min(distance, self.step_width_limit)

        # Update position
        constrained_pose.pose.position.x = current_pose.pose.position.x + dx
        constrained_pose.pose.position.y = current_pose.pose.position.y + dy
        constrained_pose.pose.position.z = target_pose.pose.position.z  # Keep original Z

        # Copy orientation
        constrained_pose.pose.orientation = target_pose.pose.orientation

        return constrained_pose

    def is_pose_valid(self, target_pose, current_pose):
        """Check if target pose is valid under bipedal constraints"""
        # Check step width constraint
        dx = target_pose.pose.position.x - current_pose.pose.position.x
        dy = target_pose.pose.position.y - current_pose.pose.position.y
        distance = math.sqrt(dx*dx + dy*dy)

            return False

        # Check step height constraint
