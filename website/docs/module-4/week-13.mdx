## Introduction to Voice-to-Action Systems from a Startup Perspective

In this final week, we'll explore the cutting-edge field of voice-to-action systems for humanoid robots. As a startup founder, understanding how to integrate natural language processing with robot control is crucial for creating intuitive human-robot interaction. Voice interfaces enable robots to understand and respond to human commands in a natural way, significantly improving usability and accessibility.

### Voice-to-Action Pipeline: From Speech to Robot Actions

The voice-to-action system consists of several components working together to convert human speech into robot actions. This pipeline typically includes:

1. **Speech Recognition**: Converting audio to text
2. **Natural Language Understanding**: Parsing and interpreting the text
3. **Action Mapping**: Translating commands to robot actions
4. **Execution**: Carrying out the actions through robot control systems

### Python Snippet: LLM-to-ROS 2 Mapping Logic

Here's a complete implementation of the LLM-based cognitive planning system that maps natural language commands to ROS 2 actions:

```python
import openai
import json
import asyncio
from typing import Dict, List, Tuple, Any
from dataclasses import dataclass
from enum import Enum
import numpy as np

class ActionType(Enum):
    """
    Enum for different types of robot actions
    """
    MOVE_TO = "move_to"
    GRASP = "grasp"
    PLACE = "place"
    FOLLOW = "follow"
    SPEAK = "speak"
    SEARCH = "search"
    GREET = "greet"

@dataclass
class RobotAction:
    """
    Data class representing a robot action
    """
    action_type: ActionType
    parameters: Dict[str, Any]
    priority: int = 1  # Higher priority actions are executed first

class LLMBridge:
    """
    Bridge between LLMs and ROS 2 robot actions
    """
    def __init__(self, api_key: str, model: str = "gpt-4o"):
        """
        Initialize LLM bridge
        :param api_key: OpenAI API key
        :param model: GPT model to use
        """
        openai.api_key = api_key
        self.model = model
        self.action_history = []

    async def interpret_command(self, command: str, environment_state: Dict) -> List[RobotAction]:
        """
        Interpret natural language command and return action plan
        :param command: Natural language command
        :param environment_state: Current environment state
        :return: List of RobotAction objects
        """
        # Create system prompt for the LLM
        system_prompt = f"""
        You are a robot command interpreter. Your task is to convert natural language commands into structured robot actions.

        Current environment state:
        {json.dumps(environment_state, indent=2)}

        Available action types:
        - move_to: Move to a location with parameters (x, y, z coordinates)
        - grasp: Grasp an object with parameters (object_name, position)
        - place: Place an object at a location
        - follow: Follow a person or object
        - speak: Speak a message
        - search: Search for an object in the environment
        - greet: Greet a person

        Response format: Return a JSON object with an 'actions' array containing action objects.
        Each action object should have 'action_type' and 'parameters' fields.

        Example:
        Input: "Go to the kitchen and pick up the red cup"
        Output: {{
            "actions": [
                {{
                    "action_type": "move_to",
                    "parameters": {{"x": 2.0, "y": 1.0, "z": 0.0}}
                }},
                {{
                    "action_type": "grasp",
                    "parameters": {{"object_name": "red cup", "position": [2.0, 1.0, 0.8]}}
                }}
            ]
        }}
        """

        user_prompt = f"Command: {command}"

        response = await openai.ChatCompletion.acreate(
            model=self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=0.1,
            max_tokens=500
        )

        response_text = response.choices[0].message.content.strip()

        # Clean up response if it contains markdown formatting
        if response_text.startswith("```json"):
            response_text = response_text[7:]  # Remove ```json
        if response_text.endswith("```"):
            response_text = response_text[:-3]  # Remove ```

        try:
            parsed_response = json.loads(response_text)
            actions_data = parsed_response.get("actions", [])

            # Convert to RobotAction objects
            robot_actions = []
            for action_data in actions_data:
                action_type = ActionType(action_data["action_type"])
                parameters = action_data.get("parameters", {})
                priority = action_data.get("priority", 1)

                robot_action = RobotAction(
                    action_type=action_type,
                    parameters=parameters,
                    priority=priority
                )
                robot_actions.append(robot_action)

            # Add to history
            self.action_history.append({
                "command": command,
                "environment_state": environment_state,
                "actions": robot_actions
            })

            return robot_actions

        except json.JSONDecodeError as e:
            print(f"Error parsing LLM response: {e}")
            print(f"Response text: {response_text}")
            return []

    async def generate_response(self, command: str, action_plan: List[RobotAction]) -> str:
        """
        Generate response to the user based on command and action plan
        :param command: Original command
        :param action_plan: Generated action plan
        :return: Response string
        """
        response_prompt = f"""
        You are a friendly humanoid robot. You received a command and generated an action plan.
        Respond to the user in a positive, helpful tone confirming understanding of their request.

        Command: {command}

        Action plan:
        {json.dumps([{
            "action_type": action.action_type.value,
            "parameters": action.parameters
        } for action in action_plan], indent=2)}

        Respond with acknowledgment of the command and confirmation that you will execute the plan.
        Keep the response under 100 words and maintain a professional yet approachable tone.
        """

        response = await openai.ChatCompletion.acreate(
            model=self.model,
            messages=[
                {"role": "system", "content": response_prompt},
                {"role": "user", "content": f"Command: {command}"}
            ],
            temperature=0.7,
            max_tokens=150
        )

        return response.choices[0].message.content.strip()

    async def execute_action_plan(self, action_plan: List[RobotAction]) -> bool:
        """
        Execute a sequence of robot actions
        In a real system, this would interface with ROS 2
        :param action_plan: List of RobotAction objects to execute
        :return: True if all actions completed successfully, False otherwise
        """
        for action in action_plan:
            success = await self.execute_single_action(action)
            if not success:
                print(f"Action failed: {action}")
                return False
            # Small delay between actions
            await asyncio.sleep(0.5)

        return True

    async def execute_single_action(self, action: RobotAction) -> bool:
        """
        Execute a single robot action (placeholder implementation)
        In a real system, this would call ROS 2 services/clients
        :param action: RobotAction to execute
        :return: True if successful, False otherwise
        """
        print(f"Executing action: {action.action_type.value}")
        print(f"Parameters: {action.parameters}")

        # In a real implementation, this would call ROS 2 services
        # For now, we'll simulate execution
        await asyncio.sleep(1.0)  # Simulate processing time

        return True

# Example usage
async def example_llm_integration():
    llm_bridge = LLMBridge(api_key="your-openai-api-key")

    # Example environment state
    env_state = {
        "objects": {
            "red_cup": {"position": [2.0, 1.0, 0.8], "properties": {"color": "red", "type": "cup"}},
            "blue_ball": {"position": [1.5, 0.5, 0.1], "properties": {"color": "blue", "type": "ball"}}
        },
        "locations": {
            "kitchen": {"coordinates": [2.0, 1.0, 0.0]},
            "living_room": {"coordinates": [0.0, 0.0, 0.0]},
            "bedroom": {"coordinates": [-1.0, 1.5, 0.0]}
        },
        "robot_pose": {"position": [0.0, 0.0, 0.0], "orientation": [0, 0, 0, 1]}
    }

    # Example commands
    commands = [
        "Go to the kitchen and pick up the red cup",
        "Find the blue ball in the living room",

### VLA Logic: Prompt Engineering for "Pick up the blue ball" to ROS 2 Coordinates

Vision-Language-Action (VLA) systems require sophisticated prompt engineering to effectively translate natural language commands into specific robot actions. Here's how to map "Pick up the blue ball" to ROS 2 coordinates:

```python
class VLAPromptEngineer:
    """
    Vision-Language-Action prompt engineering system
    """
    def __init__(self):
        self.object_recognition_system = ObjectRecognitionSystem()
        self.kinematics_calculator = KinematicsCalculator()

    def create_vla_prompt(self, command: str, robot_state: Dict, environment_state: Dict) -> str:
        """
        Create a VLA prompt that guides the LLM to generate specific actions
        :param command: Natural language command (e.g., "Pick up the blue ball")
        :param robot_state: Current robot state (position, joint angles, etc.)
        :param environment_state: Current environment state (objects, locations, etc.)
        :return: Formatted prompt for the LLM
        """
        prompt = f"""
        You are a Vision-Language-Action (VLA) coordinator for a humanoid robot. Your task is to interpret natural language commands and generate specific, executable actions.

        Current robot state:
        {json.dumps(robot_state, indent=2)}

        Current environment state:
        {json.dumps(environment_state, indent=2)}

        Command: "{command}"

        Analysis steps:
        1. Identify the action to perform (e.g., grasp, move, speak)
        2. Identify the target object ("blue ball")
        3. Locate the object in the environment state
        4. Calculate the required robot movements to reach the object
        5. Generate specific ROS 2 action commands

        Response format: JSON with "actions" array containing:
        - action_type: Type of action (move_to, grasp, etc.)
        - parameters: Specific parameters for the action
        - priority: Execution priority (1-10)

        For grasping tasks:
        - First, move to a position where the object is reachable
        - Then, execute the grasp action with specific coordinates
        - Consider the robot's kinematic constraints

        Example:
        Command: "Pick up the blue ball at position [1.5, 0.5, 0.1]"
        Response: {{
            "actions": [
                {{
                    "action_type": "move_to",
                    "parameters": {{"target_pose": [1.4, 0.4, 0.0]}},
                    "priority": 5
                }},
                {{
                    "action_type": "grasp",
                    "parameters": {{"object_position": [1.5, 0.5, 0.1], "gripper_width": 0.05}},
                    "priority": 6
                }}
            ]
        }}
        """

        return prompt

    def map_command_to_ros_coordinates(self, command: str, environment_state: Dict) -> List[Dict]:
        """
        Map a natural language command to specific ROS 2 coordinate actions
        Example: "Pick up the blue ball" -> [move_to coordinates, grasp action]
        """
        # Parse the command to identify the target object
        if "pick up" in command.lower() or "grasp" in command.lower() or "take" in command.lower():
            # Extract object name from command
            import re
            # Look for color + object pattern (e.g., "blue ball", "red cup")
            color_object_match = re.search(r'(blue|red|green|yellow|black|white|purple|orange)\s+(\w+)', command.lower())

            if color_object_match:
                color, obj_type = color_object_match.groups()
                target_object_name = f"{color} {obj_type}"

                # Find the object in the environment state
                for obj_name, obj_data in environment_state.get("objects", {}).items():
                    if target_object_name in obj_name.lower() or obj_name.lower() in target_object_name:
                        # Calculate approach position (slightly in front of the object)
                        obj_pos = obj_data["position"]
                        approach_pos = [
                            obj_pos[0] - 0.1,  # 10cm in front
                            obj_pos[1],
                            obj_pos[2]
                        ]

                        # Generate ROS 2 action sequence
                        actions = [
                            {
                                "action_type": "move_to",
                                "parameters": {
                                    "target_pose": approach_pos,
                                    "planner": "nav2"
                                },
                                "priority": 5
                            },
                            {
                                "action_type": "move_to",
                                "parameters": {
                                    "target_pose": obj_pos,
                                    "approach_velocity": 0.1  # Slow approach for precision
                                },
                                "priority": 6
                            },
                            {
                                "action_type": "grasp",
                                "parameters": {
                                    "object_name": obj_name,
                                    "grasp_type": "precision",
                                    "grasp_pose": obj_pos,
                                    "force_limit": 10.0  # Newtons
                                },
                                "priority": 7
                            }
                        ]

                        return actions

        # If no specific object found, return a generic response
        return [
            {
                "action_type": "speak",
                "parameters": {"message": f"Sorry, I couldn't identify the object in command: {command}"},
                "priority": 1
            }
        ]

# Example usage of VLA mapping
def example_vla_mapping():

### Integration with Module 1 (ROS 2) Concepts

The voice-to-action system integrates with the ROS 2 infrastructure established in Module 1. Here's how to implement the complete system as a ROS 2 node:

```python
import rclpy
from rclpy.node import Node
from std_msgs.msg import String
from geometry_msgs.msg import Pose, Twist
from sensor_msgs.msg import JointState, Image
from builtin_interfaces.msg import Duration
from trajectory_msgs.msg import JointTrajectory, JointTrajectoryPoint
import asyncio
import threading
from typing import Dict, List, Any

class VoiceCommandNode(Node):
    """
    ROS 2 node that integrates voice processing with robot control
    """
    def __init__(self):
        super().__init__('voice_command_node')

        # Initialize VLA components
        self.whisper_processor = WhisperVoiceProcessor(api_key="your-whisper-key")
        self.llm_bridge = LLMBridge(api_key="your-llm-key")
        self.vla_prompt_engineer = VLAPromptEngineer()

        # Publishers
        self.joint_trajectory_pub = self.create_publisher(
            JointTrajectory, '/joint_trajectory_controller/joint_trajectory', 10)
        self.cmd_vel_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        self.speech_pub = self.create_publisher(String, '/tts_input', 10)

        # Subscribers
        self.voice_command_sub = self.create_subscription(
            String, '/voice_commands', self.voice_command_callback, 10)
        self.joint_state_sub = self.create_subscription(
            JointState, '/joint_states', self.joint_state_callback, 10)
        self.camera_sub = self.create_subscription(
            Image, '/camera/rgb/image_raw', self.camera_callback, 10)

        # Internal state
        self.current_joint_states = None
        self.current_camera_image = None
        self.current_robot_pose = {"position": [0.0, 0.0, 0.0], "orientation": [0, 0, 0, 1]}
        self.is_processing = False

        # Timer for system monitoring
        self.monitor_timer = self.create_timer(1.0, self.system_monitor)

        self.get_logger().info('Voice Command Node initialized')

    def voice_command_callback(self, msg: String):
        """
        Handle incoming voice commands
        """
        command = msg.data
        self.get_logger().info(f'Received voice command: {command}')

        if not self.is_processing:
            # Process command in a separate thread to avoid blocking
            thread = threading.Thread(
                target=self.process_voice_command_thread,
                args=(command,)
            )
            thread.start()
        else:
            self.get_logger().warn('System busy, command queued')

    def process_voice_command_thread(self, command: str):
        """
        Process voice command in a separate thread
        """
        try:
            self.is_processing = True
            self.get_logger().info(f'Processing command: {command}')

            # Get current environment state
            env_state = self.get_current_environment_state()

            # Process with VLA pipeline
            loop = asyncio.new_event_loop()
            asyncio.set_event_loop(loop)

            # Interpret command with LLM
            action_plan = loop.run_until_complete(
                self.llm_bridge.interpret_command(command, env_state)
            )

            if action_plan:
                # Execute the action plan
                success = loop.run_until_complete(
                    self.llm_bridge.execute_action_plan(action_plan)
                )

                if success:
                    self.get_logger().info('Command executed successfully')
                    # Generate and publish response
                    response = loop.run_until_complete(
                        self.llm_bridge.generate_response(command, action_plan)
                    )
                    response_msg = String()
                    response_msg.data = response
                    self.speech_pub.publish(response_msg)
                else:
                    self.get_logger().error('Command execution failed')
                    error_msg = String()
                    error_msg.data = f"Sorry, I couldn't complete the command: {command}"
                    self.speech_pub.publish(error_msg)
            else:
                self.get_logger().error('No action plan generated from command')
                error_msg = String()
                error_msg.data = f"Sorry, I couldn't understand the command: {command}"
                self.speech_pub.publish(error_msg)

        except Exception as e:
            self.get_logger().error(f'Error processing command: {str(e)}')
            error_msg = String()
            error_msg.data = f"Sorry, I encountered an error: {str(e)}"
            self.speech_pub.publish(error_msg)
        finally:
            self.is_processing = False

    def joint_state_callback(self, msg: JointState):
        """
        Handle joint state feedback
        """
        self.current_joint_states = msg

    def camera_callback(self, msg: Image):
        """
        Handle camera images for vision processing
        """
        self.current_camera_image = msg

    def get_current_environment_state(self) -> Dict:
        """
        Get the current environment state including objects, locations, and robot pose
        """
        # In a real implementation, this would integrate with perception systems
        # For this example, we'll return a mock environment state
        return {
            "objects": {
                "blue_ball": {
                    "position": [1.5, 0.5, 0.1],
                    "properties": {"color": "blue", "type": "ball", "confidence": 0.95}
                },
                "red_cup": {
                    "position": [2.0, 1.0, 0.8],
                    "properties": {"color": "red", "type": "cup", "confidence": 0.92}
                }
            },
            "locations": {
                "kitchen": {"coordinates": [2.0, 1.0, 0.0]},
                "living_room": {"coordinates": [0.0, 0.0, 0.0]},
                "bedroom": {"coordinates": [-1.0, 1.5, 0.0]}
            },
            "robot_pose": self.current_robot_pose,
            "navigation_map": {
                "obstacles": [],
                "free_space": [[x, y] for x in range(-5, 6) for y in range(-5, 6)]
            }
        }

    def system_monitor(self):
        """
        Monitor system status and health
        """
        status = {
            "voice_system": "active" if self.current_joint_states else "inactive",
            "vision_system": "active" if self.current_camera_image else "inactive",
            "processing": self.is_processing,
            "robot_pose": self.current_robot_pose
        }

        self.get_logger().debug(f'System status: {status}')

def main(args=None):
    rclpy.init(args=args)
    voice_node = VoiceCommandNode()

    try:
        rclpy.spin(voice_node)
    except KeyboardInterrupt:
        voice_node.get_logger().info('Shutting down Voice Command Node')
    finally:
        voice_node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
```
